namespace = sdf

# Test shortcuts for a throwaway game (Test run 1). Play a nomad empire; run devtools.5 (or meet an
# empire) first: the issuer needs communications with us. Select a star or a planet, then
#   event sdf.90
# An AI empire that can issue contracts loses a corvette there, leaving debris it owns
# (destroy_and_spawn_debris_for, effects.log); a day later (sdf.91) it issues the Scan Debris
# contract there, open to any nomad (accept it before an AI nomad does). Each step logs an
# "SDF test" line to game.log. One test at a time: a second sdf.90 before sdf.91 is refused.
# Run 1 (2026-11-23) used a science ship: the debris was gone by the next day and the contract
# aborted on issue (MESSAGE_CONTRACT_ABORTED, dated the issue day). Vanilla destroys warships with
# it (first_contact_events.txt); sdf.91 now logs whether the issuer sees the debris.
# Run 2: the issuer saw it, and both contracts still aborted on issue. The cause was the
# `target = <us>`: a targeted contract's scope is the target (the save's mission scope: country 0,
# from = the issuer), and issued_abort_trigger opens with `is_nomadic = yes`. Vanilla never
# targets contract_scan_debris; now neither do we.
# Run 3: issued untargeted, then gone by the next autosave without a message (an untargeted contract
# that aborts before acceptance tells no one), the debris too. sdf.92 now logs daily for five days
# whether the issuer still sees the debris (issued_abort_trigger aborts the day it does not).
# Run 4: seen on the issue day, gone the next - not "unavailable" but gone from the save (debris
# lasts DEBRIS_DAYS = 1800, analysing takes 90). Suspect: our own arkship sat in that system with
# the debris policy on Scavenge. sdf.93 picks a system without our fleets; sdf.90 warns.


# Selected star or planet: in its system.
event = {
	id = sdf.90
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { sdf_test_can_start = no }
			sdf_test_refused = yes
		}
		else = {
			solar_system = { save_global_event_target_as = sdf_test_system }
			sdf_test_wreck = yes
		}
	}
}

# Nothing selected: a system next to our capital (the arkship) without our fleets - our nomads
# would scavenge the debris (run 4). One jump for the science ship.
country_event = {
	id = sdf.93
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { sdf_test_can_start = no }
			sdf_test_refused = yes
		}
		else_if = {
			limit = {
				capital_scope.solar_system = {
					NOT = {
						any_neighbor_system = {
							NOT = { any_fleet_in_system = { owner = { is_same_value = root } } }
						}
					}
				}
			}
			log = "SDF test: refused - every system next to our capital has a fleet of ours"
		}
		else = {
			capital_scope.solar_system = {
				random_neighbor_system = {
					limit = { NOT = { any_fleet_in_system = { owner = { is_same_value = root } } } }
					save_global_event_target_as = sdf_test_system
				}
			}
			sdf_test_wreck = yes
		}
	}
}

# this = the issuer, a day after sdf.90.
country_event = {
	id = sdf.91
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { any_available_debris = { solar_system = { is_same_value = event_target:sdf_test_system } } }
			log = "SDF test: [Root.GetName] sees the debris - issuing"
		}
		else = {
			log = "SDF test: [Root.GetName] sees no available debris there - the contract will abort"
		}
		issue_contract = {
			contract = contract_scan_debris
			location = event_target:sdf_test_system
		}
		event_target:sdf_test_system = {
			log = "SDF test: [sdf_test_issuer.GetName] issued Scan Debris in [This.GetName]"
		}
		set_variable = { which = sdf_test_days value = 0 }
		country_event = { id = sdf.92 days = 1 }
	}
}

# this = the issuer: daily for five days after issuing, does it still see the debris?
country_event = {
	id = sdf.92
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		change_variable = { which = sdf_test_days value = 1 }
		if = {
			limit = { any_available_debris = { solar_system = { is_same_value = event_target:sdf_test_system } } }
			log = "SDF test: [GetDate] day [This.sdf_test_days] after issuing - [Root.GetName] still sees the debris"
		}
		else = {
			log = "SDF test: [GetDate] day [This.sdf_test_days] after issuing - [Root.GetName] sees no debris: the contract aborts"
		}
		if = {
			limit = { check_variable = { which = sdf_test_days value < 5 } }
			country_event = { id = sdf.92 days = 1 }
		}
		else = {
			clear_variable = sdf_test_days
			clear_global_event_target = sdf_test_system
			clear_global_event_target = sdf_test_operator
			clear_global_event_target = sdf_test_issuer
			clear_global_event_target = sdf_test_victim
			remove_global_flag = sdf_test_pending
		}
	}
}
